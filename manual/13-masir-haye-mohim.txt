# مسیرهای مهم و فایل‌های پیکربندی VortexL2
# این فایل مسیرهای مهم فایل‌های کانفیگ و لاگ را نشان می‌دهد

# ============================================================================
# فایل‌های کانفیگ HAProxy
# ============================================================================

# فایل کانفیگ اصلی HAProxy (توسط VortexL2 مدیریت می‌شود)
/etc/haproxy/haproxy.cfg

# فایل پشتیبان کانفیگ HAProxy (قبل از اولین تغییر ایجاد می‌شود)
/etc/haproxy/haproxy.cfg.bak

# Socket آمار HAProxy (برای مشاهده آمار و مدیریت)
/var/run/haproxy.sock

# PID فایل HAProxy
/var/run/haproxy.pid


# ============================================================================
# فایل‌های کانفیگ VortexL2
# ============================================================================

# پوشه اصلی کانفیگ VortexL2
/etc/vortexl2/

# پوشه کانفیگ تونل‌ها (هر تونل یک فایل YAML دارد)
/etc/vortexl2/tunnels/

# مثال فایل کانفیگ یک تونل
/etc/vortexl2/tunnels/tunnel1.yaml


# ============================================================================
# فایل‌های سرویس systemd
# ============================================================================

# سرویس راه‌اندازی تونل‌ها
/etc/systemd/system/vortexl2-tunnel.service

# سرویس مدیریت Port Forwarding
/etc/systemd/system/vortexl2-forward-daemon.service


# ============================================================================
# فایل‌های اجرایی
# ============================================================================

# دستور اصلی VortexL2
/usr/local/bin/vortexl2

# پوشه نصب VortexL2
/opt/vortexl2/


# ============================================================================
# فایل‌های لاگ
# ============================================================================

# لاگ سرویس تونل
journalctl -u vortexl2-tunnel.service

# لاگ سرویس Port Forwarding
journalctl -u vortexl2-forward-daemon.service

# لاگ HAProxy
journalctl -u haproxy

# پوشه لاگ‌های VortexL2
/var/log/vortexl2/

# فایل لاگ Forward Daemon
/var/log/vortexl2/forward-daemon.log


# ============================================================================
# فایل‌های ماژول‌های کرنل
# ============================================================================

# فایل لود خودکار ماژول‌های L2TP در بوت
/etc/modules-load.d/vortexl2.conf


# ============================================================================
# دستورات مفید برای مشاهده و ویرایش
# ============================================================================

# مشاهده کانفیگ HAProxy
cat /etc/haproxy/haproxy.cfg

# ویرایش کانفیگ HAProxy (توجه: VortexL2 این فایل را مدیریت می‌کند)
nano /etc/haproxy/haproxy.cfg

# مشاهده لیست تونل‌های کانفیگ شده
ls -la /etc/vortexl2/tunnels/

# مشاهده کانفیگ یک تونل خاص
cat /etc/vortexl2/tunnels/tunnel1.yaml

# مشاهده لاگ‌های real-time
journalctl -u vortexl2-tunnel.service -f
journalctl -u haproxy -f

# بررسی وضعیت سرویس‌ها
systemctl status vortexl2-tunnel.service
systemctl status vortexl2-forward-daemon.service
systemctl status haproxy
