# ساخت چندین تونل L2TPv3 همزمان
# این فایل دستورات ساخت چندین تونل با شناسه‌های مختلف را نشان می‌دهد
# هر تونل باید tunnel_id و session_id منحصر به فرد داشته باشد

# تونل 1: l2tpeth0
# شناسه‌ها: tunnel_id=1000, session_id=10
ip l2tp add tunnel tunnel_id 1000 peer_tunnel_id 2000 encap ip local 1.2.3.4 remote 5.6.7.8
ip l2tp add session tunnel_id 1000 session_id 10 peer_session_id 20
ip link set l2tpeth0 up
ip addr add 10.30.30.1/30 dev l2tpeth0

# تونل 2: l2tpeth1
# شناسه‌ها: tunnel_id=1100, session_id=11
ip l2tp add tunnel tunnel_id 1100 peer_tunnel_id 2100 encap ip local 1.2.3.4 remote 6.7.8.9
ip l2tp add session tunnel_id 1100 session_id 11 peer_session_id 21
ip link set l2tpeth1 up
ip addr add 10.30.31.1/30 dev l2tpeth1

# تونل 3: l2tpeth2
# شناسه‌ها: tunnel_id=1200, session_id=12
ip l2tp add tunnel tunnel_id 1200 peer_tunnel_id 2200 encap ip local 1.2.3.4 remote 7.8.9.10
ip l2tp add session tunnel_id 1200 session_id 12 peer_session_id 22
ip link set l2tpeth2 up
ip addr add 10.30.32.1/30 dev l2tpeth2

# بررسی تمام تونل‌ها
ip l2tp show tunnel
ip l2tp show session

# بررسی تمام اینترفیس‌های تونل
ip addr show | grep l2tpeth

# تست اتصال برای هر تونل
ping -I l2tpeth0 10.30.30.2
ping -I l2tpeth1 10.30.31.2
ping -I l2tpeth2 10.30.32.2

# حذف تونل‌های متعدد
# ابتدا سشن‌ها را حذف کنید
ip l2tp del session tunnel_id 1000 session_id 10
ip l2tp del session tunnel_id 1100 session_id 11
ip l2tp del session tunnel_id 1200 session_id 12

# سپس تونل‌ها را حذف کنید
ip l2tp del tunnel tunnel_id 1000
ip l2tp del tunnel tunnel_id 1100
ip l2tp del tunnel tunnel_id 1200
