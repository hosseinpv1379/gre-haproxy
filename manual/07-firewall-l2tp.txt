# تنظیم فایروال برای L2TPv3
# این فایل دستورات تنظیم فایروال برای اجازه عبور ترافیک L2TP را نشان می‌دهد
# L2TPv3 از پروتکل IP با شماره 115 استفاده می‌کند

# روش 1: استفاده از iptables
# اجازه دادن به پروتکل L2TP (IP protocol 115)
iptables -A INPUT -p 115 -j ACCEPT
iptables -A OUTPUT -p 115 -j ACCEPT

# ذخیره قوانین iptables برای بوت
# برای Ubuntu/Debian:
iptables-save > /etc/iptables/rules.v4

# یا برای سیستم‌های دیگر:
# iptables-save > /etc/iptables.rules

# روش 2: استفاده از nftables (روش مدرن‌تر)
# ایجاد جدول برای فیلتر L2TP
nft create table inet vortexl2_filter

# ایجاد chain برای INPUT
nft create chain inet vortexl2_filter input { type filter hook input priority 0 \; }

# ایجاد chain برای OUTPUT
nft create chain inet vortexl2_filter output { type filter hook output priority 0 \; }

# اجازه دادن به پروتکل L2TP
nft add rule inet vortexl2_filter input ip protocol 115 accept
nft add rule inet vortexl2_filter output ip protocol 115 accept

# ذخیره قوانین nftables
nft list ruleset > /etc/nftables.conf

# روش 3: استفاده از ufw (اگر از ufw استفاده می‌کنید)
# ufw به صورت مستقیم از IP protocol پشتیبانی نمی‌کند
# باید از iptables یا nftables استفاده کنید

# بررسی قوانین فایروال
# برای iptables:
iptables -L -n -v

# برای nftables:
nft list ruleset

# تست اتصال L2TP
# اگر فایروال درست تنظیم شده باشد، تونل باید کار کند
ping 10.30.30.2
