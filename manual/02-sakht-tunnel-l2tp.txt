# ساخت تونل L2TPv3
# این فایل دستورات ساخت تونل و سشن L2TP را نشان می‌دهد
# برای هر تونل باید tunnel_id و session_id منحصر به فرد استفاده کنید

# مثال: ساخت تونل برای سمت IRAN
# پارامترها:
# - tunnel_id: شناسه تونل محلی (مثلاً 1000)
# - peer_tunnel_id: شناسه تونل ریموت (مثلاً 2000)
# - local: IP عمومی سرور محلی (مثلاً 1.2.3.4)
# - remote: IP عمومی سرور ریموت (مثلاً 5.6.7.8)

# مرحله 1: ساخت تونل L2TP
ip l2tp add tunnel tunnel_id 1000 peer_tunnel_id 2000 encap ip local 1.2.3.4 remote 5.6.7.8

# مرحله 2: ساخت سشن در تونل
# پارامترها:
# - tunnel_id: همان شناسه تونل (1000)
# - session_id: شناسه سشن محلی (مثلاً 10)
# - peer_session_id: شناسه سشن ریموت (مثلاً 20)
ip l2tp add session tunnel_id 1000 session_id 10 peer_session_id 20

# مرحله 3: بالا آوردن اینترفیس تونل
# اینترفیس به صورت خودکار با نام l2tpeth0 ایجاد می‌شود
ip link set l2tpeth0 up

# مرحله 4: اختصاص IP به اینترفیس تونل
# برای سمت IRAN معمولاً 10.30.30.1/30 استفاده می‌شود
ip addr add 10.30.30.1/30 dev l2tpeth0

# بررسی وضعیت تونل
ip l2tp show tunnel
ip l2tp show session
ip addr show l2tpeth0

# تست اتصال از طریق تونل
ping 10.30.30.2
