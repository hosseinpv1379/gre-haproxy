# پاک‌سازی و حذف کامل تنظیمات
# این فایل دستورات حذف تمام تونل‌ها، سرویس‌ها و تنظیمات را نشان می‌دهد

# مرحله 1: توقف سرویس‌ها
systemctl stop vortexl2-tunnel.service
systemctl stop haproxy
systemctl stop vortexl2-forward-daemon.service 2>/dev/null || true

# غیرفعال کردن سرویس‌ها از بوت
systemctl disable vortexl2-tunnel.service
systemctl disable haproxy
systemctl disable vortexl2-forward-daemon.service 2>/dev/null || true

# مرحله 2: حذف تمام تونل‌ها و سشن‌ها
# ابتدا لیست تمام تونل‌ها را بگیرید
ip l2tp show tunnel
ip l2tp show session

# برای هر تونل، ابتدا سشن را حذف کنید
# مثال برای tunnel_id 1000:
ip l2tp del session tunnel_id 1000 session_id 10

# سپس تونل را حذف کنید
ip l2tp del tunnel tunnel_id 1000

# تکرار برای تمام تونل‌های دیگر

# مرحله 3: حذف IP از اینترفیس‌ها
ip addr del 10.30.30.1/30 dev l2tpeth0 2>/dev/null || true
ip addr del 10.30.31.1/30 dev l2tpeth1 2>/dev/null || true
ip addr del 10.30.32.1/30 dev l2tpeth2 2>/dev/null || true

# پایین آوردن اینترفیس‌ها
ip link set l2tpeth0 down 2>/dev/null || true
ip link set l2tpeth1 down 2>/dev/null || true
ip link set l2tpeth2 down 2>/dev/null || true

# مرحله 4: بازگرداندن تنظیمات HAProxy به حالت اولیه
if [ -f /etc/haproxy/haproxy.cfg.bak ]; then
    cp /etc/haproxy/haproxy.cfg.bak /etc/haproxy/haproxy.cfg
fi

# مرحله 5: حذف فایل‌های سرویس systemd
rm -f /etc/systemd/system/vortexl2-tunnel.service
rm -f /etc/systemd/system/vortexl2-forward-daemon.service

# حذف اسکریپت‌های اجرایی
rm -f /usr/local/bin/vortexl2-tunnel-setup.sh
rm -f /usr/local/bin/vortexl2

# مرحله 6: حذف فایل‌های پیکربندی
rm -rf /etc/vortexl2
rm -rf /var/lib/vortexl2
rm -rf /var/log/vortexl2

# مرحله 7: حذف فایل ماژول‌های کرنل (اختیاری)
rm -f /etc/modules-load.d/vortexl2.conf

# مرحله 8: بارگذاری مجدد systemd
systemctl daemon-reload

# مرحله 9: حذف قوانین فایروال (اگر اضافه کرده بودید)
# برای iptables:
iptables -D INPUT -p 115 -j ACCEPT 2>/dev/null || true
iptables -D OUTPUT -p 115 -j ACCEPT 2>/dev/null || true

# برای nftables:
nft delete table inet vortexl2_filter 2>/dev/null || true

# مرحله 10: بررسی نهایی
# بررسی تونل‌ها
ip l2tp show tunnel
ip l2tp show session

# بررسی اینترفیس‌ها
ip link show | grep l2tpeth

# بررسی سرویس‌ها
systemctl list-units | grep vortexl2

# بررسی فایل‌های پیکربندی
ls -la /etc/vortexl2 2>/dev/null || echo "پوشه /etc/vortexl2 حذف شده است"
